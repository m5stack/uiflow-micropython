# LoRa-1262 Module

`LoRa1262Module` provides LoRa communication through the Module13.2
LoRa-1262 radio. The radio uses the module M-Bus SPI bus and an M5IOE1
controller on I2C for reset, antenna-switch, and power control.

Support the following product:

    LoRa-1262 Module

The constructor accepts an initialized `machine.I2C` bus. When `i2c` is
omitted, the module uses `module.mbus.i2c1`. The M5IOE1 address must be in
the range `0x71` through `0x74`.

> Note: Attach a suitable antenna before transmitting. The supported RF frequency
> range is 850000--930000 kHz.
## MicroPython Example

#### Sender

This example initializes `LoRa1262Module` with the default M-Bus I2C bus,
sends a message once per second, and displays the send timestamp.

```python
import os, sys, io
import M5
from M5 import *
from module import LoRa1262Module
import time

title0 = None
label_t = None
label_tx = None
label_ts = None
label_time = None
lora1262_0 = None

count = None
message = None

def setup():
    global title0, label_t, label_tx, label_ts, label_time, lora1262_0, count, message

    M5.begin()
    Widgets.setRotation(1)
    Widgets.fillScreen(0x222222)
    title0 = Widgets.Title("LoRa1262 Module Tx", 3, 0xFFFFFF, 0x0000FF, Widgets.FONTS.Montserrat18)
    label_t = Widgets.Label("Send:", 5, 50, 1.0, 0xFFFFFF, 0x222222, Widgets.FONTS.Montserrat18)
    label_tx = Widgets.Label("hello", 65, 50, 1.0, 0xFFFFFF, 0x222222, Widgets.FONTS.Montserrat18)
    label_ts = Widgets.Label(
        "timestamp:", 5, 150, 1.0, 0xFFFFFF, 0x222222, Widgets.FONTS.Montserrat18
    )
    label_time = Widgets.Label("1", 118, 150, 1.0, 0xFFFFFF, 0x222222, Widgets.FONTS.Montserrat18)

    count = 0
    lora1262_0 = LoRa1262Module(pin_cs=0, pin_busy=2, pin_irq=14, address=0x74)

def loop():
    global title0, label_t, label_tx, label_ts, label_time, lora1262_0, count, message
    M5.update()
    message = str("Module LoRa1262 ") + str(count)
    lora1262_0.send(message, None)
    label_tx.setText(str(message))
    label_time.setText(str(time.ticks_ms()))
    count = count + 1
    time.sleep(1)

if __name__ == "__main__":
    try:
        setup()
        while True:
            loop()
    except (Exception, KeyboardInterrupt) as e:
        try:
            from utility import print_error_msg

            print_error_msg(e)
        except ImportError:
            print("please update to latest firmware")
```

#### Receiver

This example registers a receive callback, starts continuous reception, and
displays the decoded payload, RSSI, and SNR.

```python
import os, sys, io
import M5
from M5 import *
from module import LoRa1262Module
import time

title0 = None
label_r = None
label_rx = None
label_rssi = None
label_rssi_v = None
label_snr = None
label_snr_v = None
label_t = None
label_time = None
lora1262_0 = None

lora1262_data = None
rssi = None
snr = None

def lora1262_0_receive_event(received_data):
    global \
        title0, \
        label_r, \
        label_rx, \
        label_rssi, \
        label_rssi_v, \
        label_snr, \
        label_snr_v, \
        label_t, \
        label_time, \
        lora1262_0, \
        lora1262_data, \
        rssi, \
        snr
    lora1262_data = received_data
    label_rx.setText(str(lora1262_data.decode()))
    rssi = lora1262_data.rssi
    snr = (lora1262_data.snr) / 4
    label_rssi_v.setText(str(rssi))
    label_snr_v.setText(str(snr))
    label_time.setText(str(time.ticks_ms()))

def setup():
    global \
        title0, \
        label_r, \
        label_rx, \
        label_rssi, \
        label_rssi_v, \
        label_snr, \
        label_snr_v, \
        label_t, \
        label_time, \
        lora1262_0, \
        lora1262_data, \
        rssi, \
        snr

    M5.begin()
    Widgets.setRotation(1)
    Widgets.fillScreen(0x222222)
    title0 = Widgets.Title("LoRa1262 Module Rx", 3, 0xFFFFFF, 0x0000FF, Widgets.FONTS.Montserrat18)
    label_r = Widgets.Label("Recv:", 5, 50, 1.0, 0xFFFFFF, 0x222222, Widgets.FONTS.Montserrat18)
    label_rx = Widgets.Label(" ", 65, 50, 1.0, 0xFFFFFF, 0x222222, Widgets.FONTS.Montserrat18)
    label_rssi = Widgets.Label("RSSI:", 5, 80, 1.0, 0xFFFFFF, 0x222222, Widgets.FONTS.Montserrat18)
    label_rssi_v = Widgets.Label(" ", 65, 80, 1.0, 0xFFFFFF, 0x222222, Widgets.FONTS.Montserrat18)
    label_snr = Widgets.Label("SNR:", 5, 108, 1.0, 0xFFFFFF, 0x222222, Widgets.FONTS.Montserrat18)
    label_snr_v = Widgets.Label(" ", 65, 108, 1.0, 0xFFFFFF, 0x222222, Widgets.FONTS.Montserrat18)
    label_t = Widgets.Label(
        "timestamp:", 5, 150, 1.0, 0xFFFFFF, 0x222222, Widgets.FONTS.Montserrat18
    )
    label_time = Widgets.Label("1", 118, 150, 1.0, 0xFFFFFF, 0x222222, Widgets.FONTS.Montserrat18)

    lora1262_0 = LoRa1262Module(pin_cs=0, pin_busy=2, pin_irq=14, address=0x74)
    lora1262_0.set_rx_callback(lora1262_0_receive_event)
    lora1262_0.start_recv()

def loop():
    global \
        title0, \
        label_r, \
        label_rx, \
        label_rssi, \
        label_rssi_v, \
        label_snr, \
        label_snr_v, \
        label_t, \
        label_time, \
        lora1262_0, \
        lora1262_data, \
        rssi, \
        snr
    M5.update()

if __name__ == "__main__":
    try:
        setup()
        while True:
            loop()
    except (Exception, KeyboardInterrupt) as e:
        try:
            from utility import print_error_msg

            print_error_msg(e)
        except ImportError:
            print("please update to latest firmware")
```

## API

## `LoRa1262Module`
Create a Module13.2 LoRa-1262 object.

The module uses the M-Bus SPI bus and a caller-supplied I2C bus. M5IOE1
controls SX1262 reset on G2, antenna-switch enable on G3, and LoRa power
on G5.

- Parameter `i2c`: Initialized I2C bus used to access the M5IOE1.
- Type of `i2c`: machine.I2C
- Parameter `pin_cs` (`int`): SX1262 chip-select MCU pin.
- Parameter `pin_irq` (`int`): SX1262 DIO1 MCU pin.
- Parameter `pin_busy` (`int`): SX1262 BUSY MCU pin.
- Parameter `address` (`int`): M5IOE1 I2C address, from 0x71 through 0x74.
- Parameter `freq_khz` (`int`): RF frequency in kHz, from 850000 through 930000.
- Parameter `bw` (`str`): LoRa bandwidth in kHz.
- Parameter `sf` (`int`): Spreading factor, from 6 through 12.
- Parameter `coding_rate` (`int`): Coding rate denominator, from 5 through 8.
- Parameter `preamble_len` (`int`): Preamble length, from 5 through 255 symbols.
- Parameter `syncword` (`int`): Sync word, from 1 through 255.
- Parameter `output_power` (`int`): Output power in dBm, from -9 through 22.

```python
import M5
from module import LoRa1262Module, mbus

M5.begin()
lora1262 = LoRa1262Module(mbus.i2c1, address=0x71)
```

### `set_freq`
Set the RF frequency.

- Parameter `freq_khz` (`int`): RF frequency in kHz, from 850000 through 930000.

```python
lora1262_0.set_freq(868000)
```

### `set_sf`
Set the LoRa spreading factor.

- Parameter `sf` (`int`): Spreading factor, from 6 through 12.

```python
lora1262_0.set_sf(8)
```

### `set_bw`
Set the LoRa bandwidth.

- Parameter `bw` (`str`): Bandwidth in kHz. Supported values are `"7.8"`,
    `"10.4"`, `"15.6"`, `"20.8"`, `"31.25"`, `"41.7"`,
    `"62.5"`, `"125"`, `"250"`, and `"500"`.

```python
lora1262_0.set_bw("250")
```

### `set_coding_rate`
Set the forward-error-correction coding rate.

- Parameter `coding_rate` (`int`): Coding-rate denominator, from 5 through 8.

```python
lora1262_0.set_coding_rate(8)
```

### `set_syncword`
Set the LoRa sync word.

- Parameter `syncword` (`int`): Sync word, from 1 through 255.

```python
lora1262_0.set_syncword(0x12)
```

### `set_preamble_len`
Set the LoRa preamble length.

- Parameter `preamble_len` (`int`): Preamble length, from 5 through 255 symbols.

```python
lora1262_0.set_preamble_len(12)
```

### `set_output_power`
Set the LoRa output power.

- Parameter `output_power` (`int`): Output power in dBm, from -9 through 22.

```python
lora1262_0.set_output_power(10)
```

### `send`
Send a LoRa packet.

- Parameter ` list  tuple  int  bytearray packet` (`str`): Data to send.
- Parameter `tx_at_ms` (`int`): Optional send timestamp in milliseconds.
- Returns: Send timestamp in milliseconds.
- Return type: int

```python
timestamp = lora1262_0.send("Hello LoRa1262")
```

### `recv`
Receive one LoRa packet.

- Parameter `timeout_ms` (`int`): Optional receive timeout in milliseconds.
- Parameter `rx_length` (`int`): Maximum number of bytes to receive. Default is
    `0xFF`.
- Parameter `rx_packet` (`RxPacket`): Optional packet object to reuse.
- Returns: Received packet, or `None` when the receive times out.
- Return type: RxPacket | None

```python
packet = lora1262_0.recv()
```

### `start_recv`
Start continuous LoRa reception.

```python
lora1262_0.start_recv()
```

### `set_tx_callback`
Register a callback for completed transmissions.

- Parameter `callback` (`callable`): Function called without arguments after a
    packet has been transmitted.

```python
def on_transmit():
    print("transmitted")

lora1262_0.set_tx_callback(on_transmit)
```

### `set_rx_callback`
Register a callback for received packets.

- Parameter `callback` (`callable`): Function receiving an `RxPacket` after a
    valid packet is received.

```python
def on_receive(packet):
    print(packet.decode(), packet.rssi, packet.snr / 4)

lora1262_0.set_rx_callback(on_receive)
```

### `standby`
Put the SX1262 in standby mode.

```python
lora1262_0.standby()
```

### `sleep`
Put the SX1262 in sleep mode.

```python
lora1262_0.sleep()
```

### `irq_triggered`
Return whether an SX1262 IRQ has triggered.

- Returns: `True` when an interrupt has triggered since the last
    send or receive operation started.
- Return type: bool

```python
interrupted = lora1262_0.irq_triggered()
```

### `deinit`
Put the radio to sleep and disable module power controls.

```python
lora1262_0.deinit()
```
