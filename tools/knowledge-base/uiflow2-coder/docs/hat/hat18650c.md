
# HAT 18650C

HAT 18650C V1.1 integrates an AW32257 charger and an INA226 power monitor. It
can measure battery voltage, current, and power, control battery charging, set
charge current, read charge status/fault status, and control the AW32257
Boost/OTG output.

Support the following products:

    HAT18650C

> Note: With the current hardware direction, charging current is usually negative and discharging current is usually positive.
> Charge current and voltage settings are written through AW32257 register steps, so the actual value is mapped to a supported gear rather than an arbitrary value.
## MicroPython Example

#### Charge Test

Read battery voltage and current, show charging state, and control Charge
ON/OFF with the button on StickS3.

```python
import os, sys, io
import M5
from M5 import *
from hat import HAT18650CHat
from hardware import Pin
from hardware import I2C
import time

label_title = None
label_voltage = None
label_current = None
label_iset = None
label_ctrl = None
i2c0 = None
hat_18650c_v11_0 = None

iset = None
last_time = None
bat_voltage = None
bat_current = None

def btna_was_clicked_event(state):
    global \
        label_title, \
        label_voltage, \
        label_current, \
        label_iset, \
        label_ctrl, \
        i2c0, \
        hat_18650c_v11_0, \
        iset, \
        last_time, \
        bat_voltage, \
        bat_current
    if hat_18650c_v11_0.get_charge_enable():
        hat_18650c_v11_0.set_charge_enable(False)
        label_ctrl.setText(str("Chg: OFF"))
    else:
        hat_18650c_v11_0.set_charge_enable(True)
        label_ctrl.setText(str("Chg: ON"))

def btnb_was_clicked_event(state):
    global \
        label_title, \
        label_voltage, \
        label_current, \
        label_iset, \
        label_ctrl, \
        i2c0, \
        hat_18650c_v11_0, \
        iset, \
        last_time, \
        bat_voltage, \
        bat_current
    iset = iset + 500
    if iset > 2500:
        iset = 500
    hat_18650c_v11_0.set_charge_current(iset)
    label_iset.setText(str((str("ISET: ") + str((str(iset) + str(" mA"))))))

def setup():
    global \
        label_title, \
        label_voltage, \
        label_current, \
        label_iset, \
        label_ctrl, \
        i2c0, \
        hat_18650c_v11_0, \
        iset, \
        last_time, \
        bat_voltage, \
        bat_current

    M5.begin()
    Widgets.setRotation(0)
    Widgets.fillScreen(0x000000)
    label_title = Widgets.Label(
        "Charge", 22, 5, 1.0, 0x17EE6A, 0x000000, Widgets.FONTS.Montserrat24
    )
    label_voltage = Widgets.Label(
        "Vol:", 7, 45, 1.0, 0xFFFFFF, 0x000000, Widgets.FONTS.Montserrat18
    )
    label_current = Widgets.Label(
        "Cur:", 3, 70, 1.0, 0xFFFFFF, 0x000000, Widgets.FONTS.Montserrat18
    )
    label_iset = Widgets.Label(
        "ISET: 500mA", 3, 179, 1.0, 0xFFFFFF, 0x000000, Widgets.FONTS.Montserrat18
    )
    label_ctrl = Widgets.Label(
        "CHG: ON", 26, 214, 1.0, 0xFFFFFF, 0x000000, Widgets.FONTS.Montserrat18
    )

    BtnA.setCallback(type=BtnA.CB_TYPE.WAS_CLICKED, cb=btna_was_clicked_event)
    BtnB.setCallback(type=BtnB.CB_TYPE.WAS_CLICKED, cb=btnb_was_clicked_event)

    i2c0 = I2C(0, scl=Pin(0), sda=Pin(8), freq=100000)
    hat_18650c_v11_0 = HAT18650CHat(i2c0)
    hat_18650c_v11_0.set_charge_enable(True)
    iset = 500
    hat_18650c_v11_0.set_charge_voltage(4.2)

def loop():
    global \
        label_title, \
        label_voltage, \
        label_current, \
        label_iset, \
        label_ctrl, \
        i2c0, \
        hat_18650c_v11_0, \
        iset, \
        last_time, \
        bat_voltage, \
        bat_current
    M5.update()
    if (time.ticks_diff((time.ticks_ms()), last_time)) >= 500:
        last_time = time.ticks_ms()
        bat_voltage = hat_18650c_v11_0.get_battery_voltage()
        bat_current = hat_18650c_v11_0.get_battery_current()
        print(hat_18650c_v11_0.get_battery_power())
        label_voltage.setText(str((str("Vol: ") + str((str(bat_voltage) + str("V"))))))
        label_current.setText(str((str("Cur: ") + str((str(bat_current) + str("A"))))))
        if hat_18650c_v11_0.is_charging():
            label_voltage.setColor(0x009900, 0x000000)
            label_current.setColor(0x009900, 0x000000)
        else:
            label_voltage.setColor(0xFFFFFF, 0x000000)
            label_current.setColor(0xFFFFFF, 0x000000)

if __name__ == "__main__":
    try:
        setup()
        while True:
            loop()
    except (Exception, KeyboardInterrupt) as e:
        try:
            from utility import print_error_msg

            print_error_msg(e)
        except ImportError:
            print("please update to latest firmware")
```

## **API**

#### HAT18650CHat

## `HAT18650CHat`
Create a HAT18650CHat object.

The AW32257 controls battery charging and Boost/OTG output. The INA226
measures battery bus voltage, current, and power.

- Parameter `i2c` (`I2C`): I2C bus.
- Parameter `charger_address` (`int`): AW32257 I2C address. Default is 0x6A.
- Parameter `monitor_address` (`int`): INA226 I2C address. Default is 0x41.
- Parameter `shunt_resistor` (`float`): INA226 shunt resistor value in ohms. Default is 0.01.
- Parameter `max_expected_current` (`float`): Maximum expected current for INA226 calibration in amps. Default is 10.

```python
from hardware import I2C, Pin
from hat import HAT18650CHat

i2c0 = I2C(0, scl=Pin(0), sda=Pin(8), freq=100000)
hat18650c = HAT18650CHat(i2c0)
```

### `set_charge_enable`
Enable or disable battery charging.

- Parameter `enable`: True to enable charging, False to disable charging.
- Type of `enable`: bool

```python
hat18650c.set_charge_enable(True)
hat18650c.set_charge_enable(False)
```

### `get_charge_enable`
Get battery charging enable state.

- Returns: True if charging is enabled, False otherwise.
- Return type: bool

```python
charge_enable = hat18650c.get_charge_enable()
```

### `set_charge_current`
Set fast-charge current.

The target current is clamped to the HAT 18650C supported table. The
closest supported table value not greater than the target is used
internally, so the written value may be lower than the requested value.

- Parameter `current_ma`: Target charge current in mA.
- Type of `current_ma`: int

```python
hat18650c.set_charge_current(1000)
```

### `get_charge_current`
Get fast-charge current setting.

Returns the actual selected current-table value.

- Returns: Charge current setting in mA.
- Return type: int

```python
current_ma = hat18650c.get_charge_current()
```

### `set_charge_voltage`
Set charge regulation voltage.

The target voltage is clamped to the valid AW32257 range and rounded to
the nearest register step, so the written value may be only close to the
requested value.

- Parameter `voltage`: Target charge voltage in volts. Valid range is 3.50V to 4.50V.
- Type of `voltage`: float

```python
hat18650c.set_charge_voltage(4.2)
```

### `get_charge_voltage`
Get charge regulation voltage.

Returns the actual selected voltage step.

- Returns: Charge regulation voltage in volts.
- Return type: float

```python
voltage = hat18650c.get_charge_voltage()
```

### `set_termination_current`
Set charge termination current.

The target current is rounded to the nearest termination-current table
value supported by AW32257.

- Parameter `current_ma`: Target termination current in mA. Options are 62 to 496 mA.
- Type of `current_ma`: int

```python
hat18650c.set_termination_current(124)
```

### `get_termination_current`
Get charge termination current setting.

Returns the actual selected termination-current table value.

- Returns: Termination current setting in mA.
- Return type: int

```python
current_ma = hat18650c.get_termination_current()
```

### `set_safety_charge_current`
Set maximum allowed charge current.

The target current is rounded to the nearest charge-current table value
supported by AW32257.

- Parameter `current_ma`: Target safety current in mA. Valid range is 496 to 2480 mA.
- Type of `current_ma`: int

```python
hat18650c.set_safety_charge_current(1000)
```

### `get_safety_charge_current`
Get maximum allowed charge current.

Returns the actual selected safety-current table value.

- Returns: Safety current limit in mA.
- Return type: int

```python
current_ma = hat18650c.get_safety_charge_current()
```

### `set_safety_charge_voltage`
Set maximum allowed charge regulation voltage.

The target voltage is rounded to the nearest safety-voltage table value
supported by AW32257.

- Parameter `voltage`: Target safety voltage in volts. Valid range is 4.20V to 4.50V.
- Type of `voltage`: float

```python
hat18650c.set_safety_charge_voltage(4.2)
```

### `get_safety_charge_voltage`
Get maximum allowed charge regulation voltage.

Returns the actual selected safety-voltage table value.

- Returns: Safety voltage limit in volts.
- Return type: float

```python
voltage = hat18650c.get_safety_charge_voltage()
```

### `set_recharge_threshold`
Set automatic recharge threshold below VOREG.

- Parameter `threshold_mv`: Target recharge threshold in mV. Options are 50, 100, 150, and 200.
- Type of `threshold_mv`: int

```python
hat18650c.set_recharge_threshold(100)
```

### `get_recharge_threshold`
Get automatic recharge threshold below VOREG.

- Return type: int

```python
threshold_mv = hat18650c.get_recharge_threshold()
```

### `get_charge_status`
Get charger status text.

- Returns: Charge status text: `ready`, `charging`, `done`, or `fault`.
- Return type: str

```python
status = hat18650c.get_charge_status()
```

### `get_charge_status_code`
Get charger status code.

- Returns: One of `STAT_READY`, `STAT_CHARGING`, `STAT_DONE`, or `STAT_FAULT`.
- Return type: int

```python
status = hat18650c.get_charge_status_code()
```

### `is_charging`
Check whether charging is in progress.

- Returns: True if charging is in progress, False otherwise.
- Return type: bool

```python
charging = hat18650c.is_charging()
```

### `is_charge_done`
Check whether charging is complete.

- Returns: True if charging is complete, False otherwise.
- Return type: bool

```python
done = hat18650c.is_charge_done()
```

### `get_charge_fault`
Get charger fault text.

- Returns: Charge fault text.
- Return type: str

```python
fault = hat18650c.get_charge_fault()
```

### `get_charge_fault_code`
Get charger fault code.

- Returns: Charge fault code. Use `CHARGE_FAULT_*` constants for comparison.
- Return type: int

```python
fault = hat18650c.get_charge_fault_code()
if fault == HAT18650CHat.CHARGE_FAULT_NO_BATTERY:
    print("No battery")
```

### `set_boost_enable`
Enable or disable AW32257 Boost/OTG mode.

- Parameter `enable`: True to enable Boost/OTG mode, False to disable it.
- Type of `enable`: bool

```python
hat18650c.set_boost_enable(True)
hat18650c.set_boost_enable(False)
```

### `get_boost_enable`
Get AW32257 Boost/OTG enable state.

- Returns: True if Boost/OTG mode is enabled, False otherwise.
- Return type: bool

```python
boost_enable = hat18650c.get_boost_enable()
```

### `get_boost_fault`
Get Boost/OTG fault text.

- Returns: Boost/OTG fault text.
- Return type: str

```python
fault = hat18650c.get_boost_fault()
```

### `get_boost_fault_code`
Get Boost/OTG fault code.

- Returns: Boost/OTG fault code. Use `BOOST_FAULT_*` constants for comparison.
- Return type: int

```python
fault = hat18650c.get_boost_fault_code()
if fault == HAT18650CHat.BOOST_FAULT_BATTERY_LOW:
    print("Battery low")
```

### `get_battery_voltage`
Get battery bus voltage from INA226.

- Returns: Battery voltage in volts.
- Return type: float

```python
voltage = hat18650c.get_battery_voltage()
```

### `get_battery_current`
Get battery current from INA226.

With the current hardware direction, charging current is usually negative
and discharging current is usually positive.

- Returns: Battery current in amps.
- Return type: float

```python
current = hat18650c.get_battery_current()
```

### `get_battery_power`
Get battery power from INA226.

- Returns: Battery power in watts.
- Return type: float

```python
power = hat18650c.get_battery_power()
```
